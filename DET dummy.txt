// --- TIER-BASED LOT MULTIPLIER ---
// Every trade is copied. Only the lot size changes based on tier/float coverage.
double TierLotMultiplier(string tier) {
    if(tier == "Amber") return 0.10;      // Smallest accounts — 10% of master lot
    if(tier == "Amethyst") return 0.25;   // 25% of master lot

    if(tier == "Tanzanite") return 0.75;  // 75% of master lot
    if(tier == "Sapphire") return 1.00;   // 100% — matches master exactly
    if(tier == "Emerald") return 1.50;    // 150% — larger accounts
    if(tier == "Diamond") return 2.00;    // 200% — institutional scale
    if(tier == "Rhodium") return 3.00;
    if(tier == "Platinum") return 5.00;
    if(tier == "Uranium") return 10.00;
    if(tier == "Atomic") return 20.00;
    if(tier == "Nuclear") return 50.00;
    if(tier == "Solomonic") return 100.00;
    return 0.25; // Default
}

// ============================================================
// INLINE JSON PARSER
// ============================================================
string JsonGetValue(const string &json, const string &key) {
    string search = "\"" + key + "\"";
    int pos = StringFind(json, search);
    if(pos < 0) return "";
    pos = StringFind(json, ":", pos);
    if(pos < 0) return "";
    pos++;
    while(pos < StringLen(json) && (StringGetCharacter(json, pos) == ' ' || StringGetCharacter(json, pos) == '\t')) pos++;
    if(StringGetCharacter(json, pos) == '"') {
        pos++; string val = "";
        while(pos < StringLen(json) && StringGetCharacter(json, pos) != '"') { val += ShortToString(StringGetCharacter(json, pos)); pos++; }
        return val;
    } else {
        string val = "";
        while(pos < StringLen(json)) { ushort c = StringGetCharacter(json, pos); if(c == ',' || c == '}' || c == ']') break; val += ShortToString(c); pos++; }
        return val;
    }
}

string ExtractNested(const string &json, const string &key) {
    string search = "\"" + key + "\"";
    int start = StringFind(json, search);
    if(start < 0) return "";
    start = StringFind(json, "{", start);
    if(start < 0) return "";
    int depth = 1, end = start + 1;
    while(end < StringLen(json) && depth > 0) { ushort c = StringGetCharacter(json, end); if(c == '{') depth++; else if(c == '}') depth--; end++; }
    return StringSubstr(json, start, end - start);
}

// ============================================================
// AUTO SYMBOL MATCHING
// ============================================================
string AutoMatchSymbol(const string masterSymbol) {
    if(SymbolInfoDouble(masterSymbol, SYMBOL_BID) > 0) return masterSymbol;

    string base = masterSymbol;
    StringReplace(base, ".pro", ""); StringReplace(base, ".ecn", "");
    StringReplace(base, "m", ""); StringReplace(base, "x", "");
    StringReplace(base, "-", ""); StringReplace(base, "_", "");
    StringReplace(base, "/", "");

    if(base != masterSymbol && SymbolInfoDouble(base, SYMBOL_BID) > 0) return base;

    int total = SymbolsTotal(true);
    for(int i = 0; i < total; i++) {
        string sym = SymbolName(i, true);
        string symClean = sym;
        StringReplace(symClean, ".pro", ""); StringReplace(symClean, ".ecn", "");
        StringReplace(symClean, "m", ""); StringReplace(symClean, "x", "");
        StringReplace(symClean, "-", ""); StringReplace(symClean, "_", "");
        StringReplace(symClean, "/", "");
        if(StringFind(symClean, base) >= 0 || StringFind(base, symClean) >= 0) return sym;
    }
    return "";
}

void OnTimer() {
    double equity = AccountInfoDouble(ACCOUNT_EQUITY);
    g_clientFloat = equity;
    g_traderLevel = CalculateTraderLevel(equity);
    SyncWithServer(equity);
    UpdateUI();

int CalculateTraderLevel(double equity) {
    int level = 0;
    for(int i = 0; i < ArraySize(g_levels); i++)
        if(equity >= g_levels[i].minFloat) level = g_levels[i].level;
    return level;
}

// ============================================================
// SERVER SYNC
// ============================================================
void SyncWithServer(double equity) {
    string jsonPayload = StringFormat(
        "{\"licenseKey\":\"%s\",\"currentBalance\":%.2f,\"currentEquity\":%.2f}",
        LicenseKey, equity, equity
    );

    char data[], result[]; string resHeaders;
    StringToCharArray(jsonPayload, data, 0, WHOLE_ARRAY, CP_UTF8);
    ArrayResize(data, ArraySize(data) - 1);

    string headers = "Content-Type: application/json\r\n";
    int res = WebRequest("POST", Client_API_URL, headers, 5000, data, result, resHeaders);
    if(res == -1) return;

    string response = CharArrayToString(result, 0, WHOLE_ARRAY);
    string action = JsonGetValue(response, "action");
    g_userTier = JsonGetValue(response, "userTier");

    if(action == "TRADE") {
        string masterState = ExtractNested(response, "masterState");
        if(masterState != "") {
            string openTrades = JsonGetValue(masterState, "openTrades");
            ObjectSetString(0, Prefix+"P1", OBJPROP_TEXT, "Master Trades: " + openTrades);
        }

        string uiState = ExtractNested(masterState, "uiState");
        if(uiState != "") {
            string trends = ExtractNested(uiState, "trends");
            if(trends != "") {
                ObjectSetString(0, Prefix+"V_Scalpa", OBJPROP_TEXT, ": " + JsonGetValue(trends, "scalpa"));
                ObjectSetString(0, Prefix+"V_Intra",  OBJPROP_TEXT, ": " + JsonGetValue(trends, "intra"));
                ObjectSetString(0, Prefix+"V_Day",    OBJPROP_TEXT, ": " + JsonGetValue(trends, "day"));
                ObjectSetString(0, Prefix+"V_Swing",  OBJPROP_TEXT, ": " + JsonGetValue(trends, "swing"));
                ObjectSetString(0, Prefix+"V_Long",   OBJPROP_TEXT, ": " + JsonGetValue(trends, "long"));
                ObjectSetString(0, Prefix+"V_Real",   OBJPROP_TEXT, ": " + JsonGetValue(trends, "real"));
            }
            string analytics = ExtractNested(uiState, "analytics");
            if(analytics != "") {
                ObjectSetString(0, Prefix+"A_Scalpa", OBJPROP_TEXT, ": " + JsonGetValue(analytics, "scalpa"));
                ObjectSetString(0, Prefix+"A_Intra",  OBJPROP_TEXT, ": " + JsonGetValue(analytics, "intra"));
                ObjectSetString(0, Prefix+"A_Day",    OBJPROP_TEXT, ": " + JsonGetValue(analytics, "day"));
                ObjectSetString(0, Prefix+"A_Swing",  OBJPROP_TEXT, ": " + JsonGetValue(analytics, "swing"));
                ObjectSetString(0, Prefix+"A_Long",   OBJPROP_TEXT, ": " + JsonGetValue(analytics, "long"));
            }
        }

        if(AutoModeActive) CopyTradesFromMaster(response);

    } else if(action == "KILL") {
        ObjectSetString(0, Prefix+"Title", OBJPROP_TEXT, "LICENSE EXPIRED");
    } else if(action == "ZERO_HEDGE") {
        ObjectSetString(0, Prefix+"Title", OBJPROP_TEXT, "ACCOUNT LOCKED");
    }
}

// ============================================================
// TRADE COPIER — All trades copied, lot size scaled by tier
// ============================================================
void CopyTradesFromMaster(const string &response) {
    struct MasterTrade { string symbol; string type; double volume; double sl; double tp; int ticket; };
    MasterTrade masterTrades[];
    int masterCount = 0;

    int pos = StringFind(response, "\"trades\":[");
    if(pos < 0) return;
    pos += 10;

    int tradesEnd = pos, depth = 1;
    int len = StringLen(response);
    while(tradesEnd < len && depth > 0) { ushort c = StringGetCharacter(response, tradesEnd); if(c == '[') depth++; else if(c == ']') depth--; tradesEnd++; }
    string tradesSection = StringSubstr(response, pos, tradesEnd - pos - 1);

    int tp = 0, tlen = StringLen(tradesSection);
    while(tp < tlen) {
        tp = StringFind(tradesSection, "\"ticket\":", tp);
        if(tp < 0) break;
        ArrayResize(masterTrades, masterCount + 1);
        tp += 9;
        string ticketStr = "";
        while(tp < tlen) { ushort c = StringGetCharacter(tradesSection, tp); if(c >= '0' && c <= '9') { ticketStr += ShortToString(c); tp++; } else break; }
        masterTrades[masterCount].ticket = (int)StringToInteger(ticketStr);
        tp = StringFind(tradesSection, "\"symbol\":\"", tp); if(tp < 0) break; tp += 10;
        masterTrades[masterCount].symbol = "";
        while(tp < tlen && StringGetCharacter(tradesSection, tp) != '"') { masterTrades[masterCount].symbol += ShortToString(StringGetCharacter(tradesSection, tp)); tp++; }
        tp = StringFind(tradesSection, "\"type\":\"", tp); if(tp < 0) break; tp += 8;
        masterTrades[masterCount].type = (StringGetCharacter(tradesSection, tp) == 'B') ? "BUY" : "SELL";
        tp = StringFind(tradesSection, "\"volume\":", tp); if(tp < 0) break; tp += 9;
        string volStr = "";
        while(tp < tlen) { ushort c = StringGetCharacter(tradesSection, tp); if((c >= '0' && c <= '9') || c == '.') { volStr += ShortToString(c); tp++; } else break; }
        masterTrades[masterCount].volume = StringToDouble(volStr);
        tp = StringFind(tradesSection, "\"sl\":", tp);
        if(tp > 0) { tp += 5; string slStr = ""; while(tp < tlen) { ushort c = StringGetCharacter(tradesSection, tp); if((c >= '0' && c <= '9') || c == '.' || c == '-') { slStr += ShortToString(c); tp++; } else break; } masterTrades[masterCount].sl = StringToDouble(slStr); }
        tp = StringFind(tradesSection, "\"tp\":", tp);
        if(tp > 0) { tp += 5; string tpStr = ""; while(tp < tlen) { ushort c = StringGetCharacter(tradesSection, tp); if((c >= '0' && c <= '9') || c == '.' || c == '-') { tpStr += ShortToString(c); tp++; } else break; } masterTrades[masterCount].tp = StringToDouble(tpStr); }
        masterCount++;
    }

    if(masterCount == 0) return;

    double lotMult = TierLotMultiplier(g_userTier);

    // Close client positions that no longer exist on master
    for(int i = PositionsTotal() - 1; i >= 0; i--) {
        ulong localTicket = PositionGetTicket(i);
        if(!PositionSelectByTicket(localTicket)) continue;
        string comment = PositionGetString(POSITION_COMMENT);
        if(StringFind(comment, "DET_") < 0) continue;

        bool found = false;
        string localSym = PositionGetString(POSITION_SYMBOL);
        long localType = PositionGetInteger(POSITION_TYPE);
        string localTypeStr = (localType == POSITION_TYPE_BUY) ? "BUY" : "SELL";

        for(int j = 0; j < masterCount; j++) {
            string mappedSym = AutoMatchSymbol(masterTrades[j].symbol);
            if(mappedSym == localSym && masterTrades[j].type == localTypeStr) { found = true; break; }
        }
        if(!found) {
            trade.PositionClose(localTicket);
            Print("🔒 Closed ", localSym, " — no longer on master");
        }
    }

    // Open/update — every master trade gets copied
    for(int j = 0; j < masterCount; j++) {
        string mappedSym = AutoMatchSymbol(masterTrades[j].symbol);
        if(mappedSym == "") continue;
        if(SymbolInfoDouble(mappedSym, SYMBOL_BID) <= 0) continue;

        double adjustedVol = NormalizeDouble(masterTrades[j].volume * lotMult, 2);
        if(adjustedVol < 0.01) adjustedVol = 0.01;

        // Check if already exists
        bool exists = false;
        for(int i = 0; i < PositionsTotal(); i++) {
            ulong t = PositionGetTicket(i);
            if(!PositionSelectByTicket(t)) continue;
            if(StringFind(PositionGetString(POSITION_COMMENT), "DET_") < 0) continue;
            string localSym = PositionGetString(POSITION_SYMBOL);
            long localType = PositionGetInteger(POSITION_TYPE);
            string localTypeStr = (localType == POSITION_TYPE_BUY) ? "BUY" : "SELL";

            if(localSym == mappedSym && localTypeStr == masterTrades[j].type) {
                exists = true;
                // Update SL/TP if changed
                double curSL = PositionGetDouble(POSITION_SL);
                double curTP = PositionGetDouble(POSITION_TP);
                if(MathAbs(curSL - masterTrades[j].sl) > 0.00001 || MathAbs(curTP - masterTrades[j].tp) > 0.00001) {
                    if(masterTrades[j].sl > 0 || masterTrades[j].tp > 0) {
                        trade.PositionModify(t, masterTrades[j].sl, masterTrades[j].tp);
                    }
                }
                break;
            }
        }

        if(!exists) {
            double ask = SymbolInfoDouble(mappedSym, SYMBOL_ASK);
            double bid = SymbolInfoDouble(mappedSym, SYMBOL_BID);
            if(masterTrades[j].type == "BUY" && ask > 0) {
                if(trade.Buy(adjustedVol, mappedSym, ask, masterTrades[j].sl, masterTrades[j].tp, "DET_" + IntegerToString(masterTrades[j].ticket))) {
                    Print("📋 BUY ", mappedSym, " | Lot: ", adjustedVol, " (Master: ", masterTrades[j].volume, ")");
                }
            } else if(masterTrades[j].type == "SELL" && bid > 0) {
                if(trade.Sell(adjustedVol, mappedSym, bid, masterTrades[j].sl, masterTrades[j].tp, "DET_" + IntegerToString(masterTrades[j].ticket))) {
                    Print("📋 SELL ", mappedSym, " | Lot: ", adjustedVol, " (Master: ", masterTrades[j].volume, ")");
                }
            }
        }
    }
}

void OnChartEvent(const int id, const long &lparam, const double &dparam, const string &sparam) {
    if(id == CHARTEVENT_OBJECT_CLICK) {
        if(sparam == Prefix+"BtnAuto") {
            AutoModeActive = !AutoModeActive;
            string txt = AutoModeActive ? "AUTO: ON" : "AUTO: OFF";
            color bg = AutoModeActive ? UI_AccentTeal : C'46,51,68';
            color tc = AutoModeActive ? clrWhite : UI_AccentRed;
            ObjectSetString(0, Prefix+"BtnAuto", OBJPROP_TEXT, txt);
            ObjectSetInteger(0, Prefix+"BtnAuto", OBJPROP_BGCOLOR, bg);
            ObjectSetInteger(0, Prefix+"BtnAuto", OBJPROP_COLOR, tc);
            ObjectSetInteger(0, sparam, OBJPROP_STATE, false);

            ToggleCollapse();
            ObjectSetInteger(0, sparam, OBJPROP_STATE, false);
        }
        if(sparam == Prefix+"BtnBuy" || sparam == Prefix+"BtnSell")
            ObjectSetInteger(0, sparam, OBJPROP_STATE, false);
    }
}
